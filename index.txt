      // ----------------------------------------------------
      // GOOGLE APPS SCRIPT WEB APP URL'si
      // (Aşağıdaki 2. adımdaki URL'yi buraya yapıştırın)
      // ----------------------------------------------------
      const googleScriptURL = "https://script.google.com/macros/s/AKfycbytThQGVOohZZffQnOGTFbYAJrLNBUaHs8cTscwLJalmqTBMDQzwU6ErD7pvfcZHCM/exec"; 

      // Google Sheets Verilerini Tutacak Dizi
      let schools = [];

      // DOM Elementleri
      const schoolList = document.getElementById("school-list");
      const sportFilter = document.getElementById("sport-filter");
      const cityFilter = document.getElementById("city-filter");
      const districtFilter = document.getElementById("district-filter");
      const ageFilter = document.getElementById("age-filter");

      // İstatistik Elementleri
      const countSchool = document.getElementById("school-count");
      const countSport = document.getElementById("sport-count");
      const countDistrict = document.getElementById("district-count");
      const countWhatsapp = document.getElementById("whatsapp-count");

      // SVG İkonları
      const iconLoc = `<svg viewBox="0 0 24 24" fill="none" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M21 10c0 7-9 13-9 13s-9-6-9-13a9 9 0 0 1 18 0z"></path><circle cx="12" cy="10" r="3"></circle></svg>`;
      const iconAge = `<svg viewBox="0 0 24 24" fill="none" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M17 21v-2a4 4 0 0 0-4-4H5a4 4 0 0 0-4 4v2"></path><circle cx="9" cy="7" r="4"></circle><path d="M23 21v-2a4 4 0 0 0-3-3.87"></path><path d="M16 3.13a4 4 0 0 1 0 7.75"></path></svg>`;
      const iconCal = `<svg viewBox="0 0 24 24" fill="none" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><rect x="3" y="4" width="18" height="18" rx="2" ry="2"></rect><line x1="16" y1="2" x2="16" y2="6"></line><line x1="8" y1="2" x2="8" y2="6"></line><line x1="3" y1="10" x2="21" y2="10"></line></svg>`;

      // Benzersiz değerleri almak için yardımcı fonksiyon
      function getUniqueValues(data, key) {
        return [...new Set(data.map(item => item[key]).filter(val => val))].sort();
      }

      // Filtreleri Doldur
      function populateFilters() {
        sportFilter.innerHTML = '<option value="">Tüm branşlar</option>';
        ageFilter.innerHTML = '<option value="">Tüm yaşlar</option>';

        const sports = getUniqueValues(schools, 'sport');
        const ages = getUniqueValues(schools, 'age');

        sports.forEach(sport => {
          sportFilter.add(new Option(sport, sport));
        });

        ages.forEach(age => {
          ageFilter.add(new Option(age, age));
        });
      }

      let nationalData = [];

      // Türkiye Ulusal API'sinden İl ve İlçe Verilerini Yükle
      async function loadNationalData() {
        try {
          const response = await fetch("https://turkiyeapi.dev/api/v1/provinces");
          const result = await response.json();
          nationalData = result.data;
          
          nationalData.sort((a, b) => a.name.localeCompare(b.name, 'tr')).forEach(province => {
            cityFilter.add(new Option(province.name, province.name));
          });
        } catch (error) {
          console.error("Ulusal veri çekilemedi. Okulların kayıtlı olduğu iller listeleniyor.", error);
          const cities = getUniqueValues(schools, 'city');
          cities.forEach(city => {
            cityFilter.add(new Option(city, city));
          });
        }
      }

      // İl değiştiğinde İlçeleri doldur
      function handleCityChange() {
        const selectedCity = cityFilter.value;
        districtFilter.innerHTML = "";
        
        if (!selectedCity) {
          districtFilter.disabled = true;
          districtFilter.add(new Option("Önce il seçin", ""));
        } else {
          districtFilter.disabled = false;
          districtFilter.add(new Option("Tüm ilçeler", ""));
          
          const province = nationalData.find(p => p.name === selectedCity);
          if (province && province.districts) {
             province.districts.sort((a, b) => a.name.localeCompare(b.name, 'tr')).forEach(d => {
               districtFilter.add(new Option(d.name, d.name));
             });
          } else {
             const districts = [...new Set(schools.filter(s => s.city === selectedCity).map(s => s.district))].sort((a,b) => a.localeCompare(b, 'tr'));
             districts.forEach(d => {
               districtFilter.add(new Option(d, d));
             });
          }
        }
        
        filterSchools();
      }

      // İstatistikleri Güncelle
      function updateStats(data) {
        countSchool.textContent = data.length;
        countSport.textContent = getUniqueValues(data, 'sport').length;
        countDistrict.textContent = getUniqueValues(data, 'district').length;
      }

      // Okulları Ekrana Yazdır
      function renderSchools(data) {
        schoolList.innerHTML = "";

        if (data.length === 0) {
          schoolList.innerHTML = `<div class="empty-state">Seçtiğiniz filtrelere uygun okul bulunamadı. Lütfen aramayı genişletin.</div>`;
          return;
        }

        data.forEach(school => {
          let cleanPhone = String(school.phone || '').replace(/[^0-9]/g, '');
          if (cleanPhone.length === 10 && !cleanPhone.startsWith('90')) cleanPhone = '90' + cleanPhone;
          else if (cleanPhone.length === 11 && cleanPhone.startsWith('0')) cleanPhone = '90' + cleanPhone.substring(1);

          const message = encodeURIComponent(`Merhaba, SporKayıt platformundan ulaşıyorum. ${school.name} için ücretsiz deneme dersi almak istiyorum. Bilgi alabilir miyim?`);
          const whatsappUrl = `https://wa.me/${cleanPhone}?text=${message}`;

          let sportIcon = '🏅';
          const s = (school.sport || '').toLowerCase();
          if (s.includes('basketbol')) sportIcon = '🏀';
          else if (s.includes('futbol')) sportIcon = '⚽';
          else if (s.includes('voleybol')) sportIcon = '🏐';
          else if (s.includes('yüzme') || s.includes('yuzme')) sportIcon = '🏊‍♂️';
          else if (s.includes('cimnastik') || s.includes('jimnastik')) sportIcon = '🤸‍♀️';
          else if (s.includes('tenis')) sportIcon = '🎾';

          let scheduleHtml = '';
          const isWeekday = school.weekday && school.weekday.trim().toLowerCase() === 'x';
          const isWeekend = school.weekend && school.weekend.trim().toLowerCase() === 'x';
          
          if (isWeekday && isWeekend) {
            scheduleHtml = `<div class="info-pill">${iconCal} <span>Hafta İçi ve Sonu</span></div>`;
          } else if (isWeekday) {
            scheduleHtml = `<div class="info-pill">${iconCal} <span>Hafta İçi</span></div>`;
          } else if (isWeekend) {
            scheduleHtml = `<div class="info-pill">${iconCal} <span>Hafta Sonu</span></div>`;
          } else if (school.schedule) {
            scheduleHtml = `<div class="info-pill">${iconCal} <span>${school.schedule}</span></div>`;
          }

          const locationText = [school.city, school.district].filter(Boolean).join(' / ');

          let ageHtml = '';
          if (school.age) {
             ageHtml = `<div class="info-pill">${iconAge} <span>${school.age} Yaş</span></div>`;
          }

          const card = document.createElement("article");
          card.className = "school-card";
          
          card.innerHTML = `
            <div class="card-header">
              <div class="sport-tag">
                <div class="sport-icon-box">${sportIcon}</div>
                <span class="sport-name">${school.sport}</span>
              </div>
            </div>
            <h3>${school.name}</h3>
            <div class="card-pills">
              <div class="info-pill">
                ${iconLoc}
                <span>${locationText}</span>
              </div>
              ${ageHtml}
              ${scheduleHtml}
            </div>
            <a href="${whatsappUrl}" target="_blank" rel="noopener noreferrer" class="whatsapp-button">
              Ücretsiz Deneme Dersi Al
              <svg viewBox="0 0 24 24"><path d="M5 12h14M12 5l7 7-7 7"/></svg>
            </a>
          `;
          schoolList.appendChild(card);
        });
      }

      function filterSchools() {
        const selectedSport = sportFilter.value;
        const selectedCity = cityFilter.value;
        const selectedDistrict = districtFilter.value;
        const selectedAge = ageFilter.value;

        const filtered = schools.filter(school => {
          const matchSport = selectedSport === "" || school.sport === selectedSport;
          const matchCity = selectedCity === "" || school.city === selectedCity;
          const matchDistrict = selectedDistrict === "" || school.district === selectedDistrict;
          const matchAge = selectedAge === "" || school.age === selectedAge;
          
          return matchSport && matchCity && matchDistrict && matchAge;
        });

        renderSchools(filtered);
      }

      async function loadSchoolsData() {
        return new Promise((resolve, reject) => {
          const sheetId = "1CQvnSukXaT6E_A8zmOobsnOYbKKnSGeSIqqS5sQQOuM";
          const csvUrl = `https://docs.google.com/spreadsheets/d/${sheetId}/export?format=csv`;

          Papa.parse(csvUrl, {
            download: true,
            header: true,
            skipEmptyLines: true,
            complete: function(results) {
              const getVal = (row, possibleKeys) => {
                const rowKeys = Object.keys(row);
                for (let key of rowKeys) {
                  if (possibleKeys.includes(key.trim().toLowerCase())) return row[key];
                }
                return '';
              };

              const parseDate = (dateStr) => {
                if (!dateStr) return null;
                const parts = dateStr.trim().split(/[\.\-\/]/);
                if (parts.length === 3) {
                  let day = parseInt(parts[0], 10);
                  let month = parseInt(parts[1], 10) - 1;
                  let year = parseInt(parts[2], 10);
                  if (year < 31 && day > 1000) { let t = day; day = year; year = t; }
                  if (year < 100) year += 2000;
                  return new Date(year, month, day);
                }
                const d = new Date(dateStr);
                return isNaN(d.getTime()) ? null : d;
              };

              const today = new Date();
              today.setHours(0, 0, 0, 0);

              const rawSchools = results.data.map(row => {
                return {
                  name: getVal(row, ['okul', 'okul adı', 'okul adi', 'name']) || 'İsimsiz Okul',
                  sport: getVal(row, ['branş', 'brans', 'sport']),
                  city: getVal(row, ['il', 'city']),
                  district: getVal(row, ['ilçe', 'ilce', 'district']),
                  age: getVal(row, ['yaş grubu', 'yaş', 'yas', 'yaş aralığı', 'age']),
                  schedule: getVal(row, ['program', 'schedule']),
                  weekday: getVal(row, ['hafta içi', 'hafta ici']),
                  weekend: getVal(row, ['hafta sonu']),
                  pitch: getVal(row, ['tesis', 'pitch']),
                  phone: getVal(row, ['whatsapp', 'telefon', 'phone']),
                  highlight: getVal(row, ['açıklama', 'öne çıkan özellik', 'highlight']),
                  startDateStr: getVal(row, ['başlangıç', 'baslangic', 'başlangıç tarihi']),
                  endDateStr: getVal(row, ['bitiş', 'bitis', 'bitiş tarihi'])
                };
              });

              schools = rawSchools.filter(school => {
                if (school.startDateStr) {
                  const sDate = parseDate(school.startDateStr);
                  if (sDate && sDate > today) return false;
                }
                if (school.endDateStr) {
                  const eDate = parseDate(school.endDateStr);
                  if (eDate && eDate < today) return false;
                }
                return true;
              });

              resolve();
            },
            error: function(err) {
              console.error("Google Sheets okuma hatası:", err);
              reject(err);
            }
          });
        });
      }

      // WhatsApp Tıklama - Google Sheets veya Yerel Hafıza
      async function fetchTotalClicks() {
        if (!googleScriptURL) {
           let local = parseInt(localStorage.getItem('waClicks')) || 0;
           countWhatsapp.textContent = local + ' tik';
           return;
        }
        try {
           const response = await fetch(googleScriptURL);
           const data = await response.json();
           countWhatsapp.textContent = data.clicks + ' tik';
        } catch(e) {
           console.log("Sunucudan sayaç çekilemedi, yerel veri gösteriliyor.");
           let local = parseInt(localStorage.getItem('waClicks')) || 0;
           countWhatsapp.textContent = local + ' tik';
        }
      }

      schoolList.addEventListener('click', (e) => {
        const btn = e.target.closest('.whatsapp-button');
        if (btn) {
           let current = parseInt(countWhatsapp.textContent) || parseInt(localStorage.getItem('waClicks')) || 0;
           current++;
           countWhatsapp.textContent = current + ' tik';
           localStorage.setItem('waClicks', current); // Her ihtimale karşı tarayıcıda tut

           if (googleScriptURL) {
               // Excel tablosundaki sayacı artırması için arkada sessizce sinyal gönderir
               fetch(googleScriptURL + "?action=click", { method: 'GET', mode: 'no-cors' }).catch(err => {});
           }
        }
      });

      sportFilter.addEventListener("change", filterSchools);
      cityFilter.addEventListener("change", handleCityChange);
      districtFilter.addEventListener("change", filterSchools);

          await loadSchoolsData();
          await loadNationalData();
          populateFilters();
          updateStats(schools);
          renderSchools(schools);
          await fetchTotalClicks();
        } catch (error) {
          schoolList.innerHTML = `<div class="empty-state">Veriler yüklenemedi. Lütfen Google E-Tablo paylaşım ayarlarını kontrol edin.</div>`;
        }
      }

      document.addEventListener("DOMContentLoaded", init);
